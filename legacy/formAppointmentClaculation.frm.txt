Function GetFilenameFromPath(ByVal strPath As String) As String
    
    If Right$(strPath, 1) <> "\" And Len(strPath) > 0 Then
        GetFilenameFromPath = GetFilenameFromPath(Left$(strPath, Len(strPath) - 1)) + Right$(strPath, 1)
    End If
    
End Function
Function isOpen(ByVal strPath As String)

    Dim wBook As Workbook
    On Error Resume Next
    Set wBook = Workbooks(strPath)
    If wBook Is Nothing Then 'Not open
       Application.Workbooks.Open (strPath)
    End If
End Function
Sub FileBrowser()
    Dim strPath As String
    Dim wb As Workbook
    strPath = Application.GetOpenFilename(, , "Select your File")
    If strPath = "" Then Exit Sub
    isOpen (GetFilenameFromPath(strPath))
    Set wb = Application.Workbooks(GetFilenameFromPath(strPath))
End Sub

'************************************************************************
'*                                              Standard Deviation                                    *
'************************************************************************
Function fMean(arr() As String)
     Dim Sum As Double
     Dim i, k As Integer

     Sum = 0

     For i = 0 To UBound(arr())
         If arr(i) <> "" Then
            Sum = Sum + arr(i)
            k = k + 1
        End If
     Next i
 
     fMean = Sum / k

End Function

'************************************************************************
'*                                              Standard Deviation                                    *
'************************************************************************
Function fStdDev(arr() As String)
     Dim i, k As Integer
     Dim avg As Double, SumSq As Double
 
     avg = fMean(arr)
     For i = 0 To UBound(arr())
          If arr(i) <> "" Then
            SumSq = SumSq + (arr(i) - avg) ^ 2
            k = k + 1
          End If
     Next i
 
     fStdDev = Sqr(SumSq / (k - 1))

End Function
